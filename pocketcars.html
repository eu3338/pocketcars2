    function renderProducts() {

      const grid = document.getElementById('product-grid');

      const searchKeyword = document.getElementById('search-input').value.toLowerCase();


      const filtered = products.filter(item => {

        const matchesCategory = (activeCategory === 'all') || (item.category === activeCategory);

        const matchesSearch = item.name.toLowerCase().includes(searchKeyword) || item.brand.toLowerCase().includes(searchKeyword);

        return matchesCategory && matchesSearch;

      });


      if (filtered.length === 0) {

        grid.innerHTML = `

          <div class="col-span-full text-center py-12 bg-white rounded-2xl border border-gray-100">

            <p class="text-gray-400 font-bold text-lg">No diecast models found matching your search.</p>

            <button onclick="resetFilters()" class="mt-3 text-hotpink text-sm font-extrabold hover:underline">Reset Filters</button>

          </div>

        `;

        return;

      }


      grid.innerHTML = filtered.map(item => `

        <div class="bg-white rounded-2xl border border-gray-100 p-4 shadow-sm hover:shadow-md transition-all flex flex-col justify-between group">

          <div>

            <div class="relative bg-gray-50 rounded-xl p-3 mb-4 flex items-center justify-center overflow-hidden cursor-pointer" onclick="openModal(${item.id})">

            </div>

            <div class="flex items-center space-x-2 mb-1">

              <span class="text-[11px] font-extrabold text-tiffany uppercase tracking-wider">${item.brand}</span>

              <span class="text-[11px] font-bold text-gray-300">•</span>

          </div>

          <div>

            <div class="flex items-center justify-between my-3">

              <span class="text-lg font-black text-gray-900">S$${item.price.toFixed(2)}</span>

              <button onclick="openModal(${item.id})" class="text-xs font-extrabold text-tiffany hover:underline">Details</button>

            </div>

            <button onclick="addToCart(${item.id})" class="w-full bg-tiffany-50 hover:bg-tiffany border border-tiffany/30 hover:border-tiffany text-tiffany hover:text-white font-extrabold text-xs py-2.5 rounded-xl transition-all">

              Add to Cart

            </button>

          </div>

        </div>

      `).join('');

    function setCategory(cat) {

      activeCategory = cat;

      const buttons = document.querySelectorAll('.chip');

      buttons.forEach(btn => {

        btn.classList.remove('bg-hotpink', 'text-white');

        btn.classList.add('bg-white', 'text-gray-600');

      });

      event.target.classList.remove('bg-white', 'text-gray-600');

      event.target.classList.add('bg-hotpink', 'text-white');

      renderProducts();

    }


    function filterProducts() {

      renderProducts();

    }


    function resetFilters() {

      document.getElementById('search-input').value = '';

      activeCategory = 'all';

      renderProducts();

    }


    function openModal(id) {

      const item = products.find(p => p.id === id);

      if (!item) return;


      document.getElementById('modal-svg-container').innerHTML = item.svg;

      document.getElementById('modal-brand').innerText = `${item.brand} | Scale ${item.scale}`;

      document.getElementById('modal-title').innerText = item.name;

      document.getElementById('modal-price').innerText = `S$${item.price.toFixed(2)}`;

      document.getElementById('modal-specs').innerText = item.specs;

      

      const addBtn = document.getElementById('modal-add-btn');

      addBtn.onclick = function() {

        addToCart(item.id);

        closeModal();

      };


      document.getElementById('product-modal').classList.remove('hidden');

    }


    function closeModal() {

      document.getElementById('product-modal').classList.add('hidden');

    }


    function addToCart(id) {

      const item = products.find(p => p.id === id);

      const existing = cart.find(c => c.id === id);


      if (existing) {

        existing.qty += 1;

      } else {

        cart.push({ ...item, qty: 1 });

      }


      updateCart();

      toggleCart(true);

    }


    function updateQty(id, delta) {

      const item = cart.find(c => c.id === id);

      if (!item) return;


      item.qty += delta;

      if (item.qty <= 0) {

        cart = cart.filter(c => c.id !== id);

      }

      updateCart();

    }


    function updateCart() {

      const cartCount = document.getElementById('cart-count');

      const cartItems = document.getElementById('cart-items');

      const cartSubtotal = document.getElementById('cart-subtotal');

      const cartGst = document.getElementById('cart-gst');

      const cartTotal = document.getElementById('cart-total');


      const totalItems = cart.reduce((sum, item) => sum + item.qty, 0);

      cartCount.innerText = totalItems;


      if (cart.length === 0) {

        cartItems.innerHTML = `

          <div class="text-center py-16 text-gray-400">

            <svg class="w-12 h-12 mx-auto text-gray-300 mb-3" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="1.5" d="M16 11V7a4 4 0 00-8 0v4M5 9h14l1 12H4L5 9z"/></svg>

            <p class="font-bold">Your cart is currently empty.</p>

          </div>

        `;

        cartSubtotal.innerText = "S$0.00";

        cartGst.innerText = "S$0.00";

        cartTotal.innerText = "S$0.00";

        return;

      }


      cartItems.innerHTML = cart.map(item => `

        <div class="flex items-center justify-between border-b border-gray-100 pb-4">

          <div class="flex items-center space-x-3">

            <div class="w-12 h-12 flex items-center justify-center bg-gray-50 rounded-lg p-1 border border-gray-100">

              ${item.svg}

            </div>

            <div>

              <h4 class="text-xs font-extrabold text-gray-900 max-w-[150px] truncate">${item.name}</h4>

              <span class="text-xs font-bold text-hotpink">S$${item.price.toFixed(2)}</span>

            </div>

          </div>

          <div class="flex items-center space-x-2 bg-gray-100 rounded-lg p-1">

            <button onclick="updateQty(${item.id}, -1)" class="w-6 h-6 flex items-center justify-center font-bold text-gray-600 hover:bg-white rounded">-</button>

            <span class="text-xs font-bold w-4 text-center">${item.qty}</span>

            <button onclick="updateQty(${item.id}, 1)" class="w-6 h-6 flex items-center justify-center font-bold text-gray-600 hover:bg-white rounded">+</button>

          </div>

        </div>

      `).join('');


      const subtotal = cart.reduce((sum, item) => sum + (item.price * item.qty), 0);

      const gst = subtotal * 0.09;

      const total = subtotal + gst;


      cartSubtotal.innerText = `S$${subtotal.toFixed(2)}`;

      cartGst.innerText = `S$${gst.toFixed(2)}`;

      cartTotal.innerText = `S$${total.toFixed(2)}`;

    }


    function toggleCart(forceOpen = false) {

      const panel = document.getElementById('cart-panel');

      const backdrop = document.getElementById('cart-backdrop');

      const drawer = document.getElementById('cart-drawer');


      if (forceOpen || panel.classList.contains('translate-x-full')) {

        drawer.classList.remove('pointer-events-none');

        backdrop.classList.remove('hidden');

        setTimeout(() => {

          backdrop.classList.remove('opacity-0');

          panel.classList.remove('translate-x-full');

        }, 10);

      } else {

        panel.classList.add('translate-x-full');

        backdrop.classList.add('opacity-0');

        setTimeout(() => {

          backdrop.classList.add('hidden');

          drawer.classList.add('pointer-events-none');

        }, 300);

      }

    }


    function checkout() {

      if (cart.length === 0) return alert("Your cart is empty!");

      alert("Order placed successfully! In a live system, this routes to your PayNow SG or credit card checkout page.");

      cart = [];

      updateCart();

      toggleCart();

    }
